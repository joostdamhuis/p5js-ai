// Live controls: tweak these in the Controls panel const dial = dials({ size: [400, 64, 400], detail: [64, 3, 64, 1], color: '#000000', background: '#ffffff', columns: [1, 1, 12, 1], spacing: [1, 1, 3, 0.05], speed: [5, 0, 5, 0.1], ripple: [0.3, 0, 1, 0.05], spin: true, text: 'HELLO', textSize: [110, 20, 250, 1], tilt: [0, -90, 90, 1], bounce: [0, 0, 100, 1], shading: false, }); let faceTex; let lastKey = ''; function setup() { createCanvas(windowWidth, windowHeight, WEBGL); noStroke(); faceTex = createGraphics(512, 512); } // Redraw the face texture only when the text or its colours change function drawFace() { const key = dial.color + dial.background + dial.text + dial.textSize; if (key === lastKey) return; lastKey = key; faceTex.background(dial.background); faceTex.fill(dial.color); faceTex.noStroke(); faceTex.textAlign(CENTER, CENTER); faceTex.textStyle(BOLD); faceTex.textSize(dial.textSize); faceTex.text(dial.text, 256, 256); } function draw() { background(dial.background); drawFace(); // Optional soft lighting so the spheres read as 3D if (dial.shading) { ambientLight(140); directionalLight(255, 255, 255, -0.5, 0.5, -1); } // Tilt the whole grid towards or away from the viewer rotateX(radians(dial.tilt)); // Grid of spheres centred on the canvas, spaced by their size (columns 1 = a single sphere) const gap = dial.size * dial.spacing; const offset = (dial.columns - 1) / 2; // Spin speed scales with the speed control; spin toggle stops the rotation const t = frameCount * 0.01 * dial.speed; const spinAngle = dial.spin ? t : 0; texture(faceTex); for (let i = 0; i < dial.columns; i++) { for (let j = 0; j < dial.columns; j++) { push(); // Bounce moves each sphere up and down, offset by the ripple const bob = sin(t + (i + j) * dial.ripple) * dial.bounce; translate((i - offset) * gap, (j - offset) * gap + bob, 0); // Each sphere spins at its own phase so the grid ripples rotateY(spinAngle + (i + j) * dial.ripple); sphere(dial.size / 2, dial.detail, dial.detail); pop(); } } } function windowResized() { resizeCanvas(windowWidth, windowHeight); }
